cmake_minimum_required(VERSION 3.18)
project(iphone_gaussian LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(MOBILEGS_COREML "Enable CoreML provider (Apple runtime required)" OFF)
set(ORT_ROOT "" CACHE PATH "ORT distribution with include/ and lib/")
add_library(mobile_geometry SHARED native/src/mobile_geometry.cpp native/src/mobile_frontend.cpp native/src/surface_sampler.cpp)
target_include_directories(mobile_geometry PUBLIC native/include)
add_library(scene_export STATIC native/src/scene_export.cpp)
target_include_directories(scene_export PUBLIC native/include)
enable_testing()
add_executable(geometry_smoke tests/geometry_smoke.cpp)
target_link_libraries(geometry_smoke PRIVATE mobile_geometry)
add_test(NAME geometry_smoke COMMAND geometry_smoke)
add_executable(frontend_replay tests/frontend_replay.cpp)
target_link_libraries(frontend_replay PRIVATE mobile_geometry)
add_executable(scene_export_cli tests/scene_export_cli.cpp)
target_link_libraries(scene_export_cli PRIVATE scene_export)
if(ORT_ROOT)
  find_library(ORT_LIBRARY NAMES onnxruntime PATHS "${ORT_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
  add_library(mobile_onnx SHARED native/src/mobile_onnx.cpp)
  target_include_directories(mobile_onnx PUBLIC native/include PRIVATE "${ORT_ROOT}/include")
  target_link_libraries(mobile_onnx PRIVATE "${ORT_LIBRARY}")
  if(MOBILEGS_COREML)
    target_compile_definitions(mobile_onnx PRIVATE MOBILEGS_COREML=1)
  endif()
  add_library(mobile_frontend SHARED native/src/photo_pipeline.cpp native/src/preprocess.cpp)
  target_include_directories(mobile_frontend PUBLIC native/include)
  target_link_libraries(mobile_frontend PRIVATE mobile_geometry mobile_onnx)
  add_executable(photo_session_bench native/src/photo_session_bench.cpp)
  target_link_libraries(photo_session_bench PRIVATE mobile_frontend)
endif()
